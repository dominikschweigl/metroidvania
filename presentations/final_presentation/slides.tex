%% SLIDE: Item System Technical Design
\begin{frame}[fragile,shrink=45]{Inventory System: Technical Design}
	\begin{columns}[T]
		\begin{column}{0.5\textwidth}
			\small
			\textbf{Uniform Slot Addressing}
			\begin{lstlisting}
enum class SlotKind {
    Hat, Gum, Backup, Grid, Hotbar
};
struct SlotRef { SlotKind kind; int index; };
			\end{lstlisting}
			Every slot -- equipment, grid, or hotbar -- is a single \texttt{SlotRef}.
			\texttt{moveItem(from,~to)} and \texttt{interact(slot,~player)} are generic entry points regardless of which area is involved.

			\vspace{0.4em}

			\textbf{Slot-Move Validation}
			\begin{lstlisting}
bool isValidInSlot(const Item& item, SlotRef s) {
    auto home = item.equipmentSlot();
    if (s.kind == Hat || s.kind == Gum
                      || s.kind == Backup)
        return home.has_value()
            && home.value() == s.kind;
    return true; // grid + hotbar: any item
}
			\end{lstlisting}
			Validation lives in the model -- UI drag-drop and keyboard shortcuts obey the same rules.
		\end{column}

		\begin{column}{0.46\textwidth}
			\small
			\textbf{Item Self-Activation}
			\begin{lstlisting}
// Equipment (e.g. HatItem)
void activate(...) {
    inventory.moveToEquipmentSlot(ownSlot);
}
// Consumable (e.g. HealingPotionItem)
void activate(...) {
    player.heal(amount);
    inventory.clearSlot(ownSlot);
}
			\end{lstlisting}

			The inventory calls \texttt{item.activate(player, *this, slot)}. The item decides what to do -- the inventory is a passive store; \textbf{logic lives in the item}.
		\end{column}
	\end{columns}
\end{frame}
